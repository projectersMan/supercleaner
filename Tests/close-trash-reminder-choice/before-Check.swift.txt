import CleanerCore
import Foundation

/// The packaged self-check uses held continuations only; it never invokes Finder.
actor TrashEmptyingFixture {
    private(set) var calls = 0
    private(set) var capacityReads = 0
    private var waiting: [CheckedContinuation<Void, Never>] = []
    private var completion: CheckedContinuation<Void, Error>?
    private let holdCapacity: Bool
    private var capacityStarted: [CheckedContinuation<Void, Never>] = []
    private var capacityCompletion: CheckedContinuation<Void, Never>?

    init(holdCapacity: Bool = false) { self.holdCapacity = holdCapacity }

    static func report(home: URL) -> SystemScanReport {
        let log = CleanupCandidate(url: home.appendingPathComponent("Library/Logs/fixture.log"), displayName: "Fixture.log",
            category: .log, bytes: 100, risk: .safe, evidence: [],
            identity: .init(device: 1, inode: 1, size: 100, modificationTime: 1, kind: .file),
            source: .general, lastAccessDate: nil, isActive: false)
        let application = InstalledApplication(bundleURL: home.appendingPathComponent("Applications/Fixture.app"),
            bundleIdentifier: "org.example.trash-fixture", name: "Fixture", version: "1", isRunning: false, isSystem: false)
        let applicationItem = CleanupCandidate(url: application.bundleURL, displayName: "Fixture", category: .application,
            bytes: 100, risk: .review, evidence: [],
            identity: .init(device: 1, inode: 2, size: 100, modificationTime: 1, kind: .directory),
            source: .application, lastAccessDate: nil, isActive: false)
        let inventory = ApplicationInventory(applications: [application], coverage: .init(), scannedAt: Date())
        let analysis = ApplicationUninstallAnalysis(application: application,
            plan: UninstallPlan(application: application, items: [.init(candidate: applicationItem, containsUserData: false, isShared: false)]), state: .complete)
        return SystemScanReport(metadata: .init(startedAt: Date(), completedAt: Date(), rulesetVersion: "fixture"),
            modules: [.general: .init(module: .general, state: .completed), .applications: .init(module: .applications, state: .completed)],
            generalCandidates: [log], applicationInventory: inventory,
            applicationUninstallInventory: .init(applicationInventory: inventory, analyses: [analysis]),
            diskUsageTree: .init(roots: [], largeFiles: [], coverage: .init(), scannedAt: Date()), largeFileCandidates: [], agentAssets: [])
    }

    func execute() async throws {
        calls += 1
        let started = waiting
        waiting.removeAll()
        started.forEach { $0.resume() }
        try await withCheckedThrowingContinuation { completion = $0 }
    }

    func waitUntilStarted() async {
        if calls > 0 { return }
        await withCheckedContinuation { waiting.append($0) }
    }

    func finish(_ result: Result<Void, Error>) {
        completion!.resume(with: result)
        completion = nil
    }

    func readCapacity(at directory: URL) async -> DiskCapacityReadResult {
        capacityReads += 1
        let waiting = capacityStarted
        capacityStarted.removeAll()
        waiting.forEach { $0.resume() }
        if holdCapacity { await withCheckedContinuation { capacityCompletion = $0 } }
        return .available(DiskCapacitySnapshot(volumeURL: directory, totalBytes: 100, availableBytes: 50, recordedAt: Date())!)
    }

    func waitUntilCapacityRead() async {
        if capacityReads > 0 { return }
        await withCheckedContinuation { capacityStarted.append($0) }
    }

    func finishCapacityRead() {
        capacityCompletion!.resume()
        capacityCompletion = nil
    }
}

@MainActor final class TrashEmptyingPresentationFixture {
    let approved: Bool
    private(set) var confirmationCount = 0
    private(set) var failures: [String] = []
    var checkConfirmation: (@MainActor () -> Bool)?
    private(set) var confirmationWasBusy = false
    private(set) var terminationReplies: [Bool] = []

    init(approved: Bool) { self.approved = approved }

    func confirm() -> Bool {
        confirmationCount += 1
        if let checkConfirmation { confirmationWasBusy = checkConfirmation() }
        return approved
    }

    func reportFailure(_ message: String) { failures.append(message) }
    func replyToTermination(_ allowed: Bool) { terminationReplies.append(allowed) }
}

struct TrashEmptyingCheckFailure: Error, CustomStringConvertible {
    let description: String
}

extension AppModel {
    static func checkTrashEmptying() async throws {
        let domain = "SuperCleaner.TrashEmptyingCheck.\(UUID().uuidString)"
        let preferences = UserDefaults(suiteName: domain)!
        defer { preferences.removePersistentDomain(forName: domain) }
        try checkClosedWindowTrashResultGate(preferences: preferences)
        let cancelled = TrashEmptyingPresentationFixture(approved: false)
        let cancelledExecution = TrashEmptyingFixture()
        let cancelledModel = makeTrashEmptyingCheckModel(fixture: cancelledExecution)
        let cancelledController = TrashReminderController(preferences: preferences,
            confirmEmptyTrash: cancelled.confirm, emptyTrash: { try await cancelledExecution.execute() }, reportFailure: cancelled.reportFailure)
        cancelled.checkConfirmation = { [weak cancelledController, weak cancelledModel] in
            guard let cancelledController, let cancelledModel else { return false }
            return !cancelledController.isEmptyingTrash && cancelledModel.trashEmptyingInProgress
                && !cancelledController.canEmptyTrash(model: cancelledModel) && !cancelledModel.canExecuteSelection
        }
        let cancellation = cancelledController.requestEmptyTrash(model: cancelledModel)
        guard cancelled.confirmationCount == 1, cancelled.confirmationWasBusy, await cancelledExecution.calls == 0,
              await cancelledExecution.capacityReads == 0, !cancelledController.isEmptyingTrash,
              !cancelledModel.trashEmptyingInProgress, cancelledController.shouldRemind(cleanedBytes: 1_024),
              !cancelledController.shouldRemind(cleanedBytes: 0), await cancellation.value == false else {
            throw TrashEmptyingCheckFailure(description: "confirmation cancellation must not execute or acknowledge cleanup")
        }
        print("PASS: trash confirmation cancellation preserves reminder and never executes")

        let outcomes: [Result<Void, Error>] = [.success(()), .failure(ProcessFailure.timedOut),
            .failure(ProcessFailure.cancelled), .failure(FinderTrashEmptyingError.failed(status: 1, message: "Not authorized (-1743)")),
            .failure(FinderTrashEmptyingError.unconfirmed)]
        for (index, outcome) in outcomes.enumerated() {
            let execution = TrashEmptyingFixture()
            let presentation = TrashEmptyingPresentationFixture(approved: true)
            let model = makeTrashEmptyingCheckModel(fixture: execution)
            let controller = TrashReminderController(preferences: preferences,
                confirmEmptyTrash: presentation.confirm, emptyTrash: { try await execution.execute() }, reportFailure: presentation.reportFailure)
            guard model.canRunOneClickCleanup, model.canExecuteSelection, model.canExecuteUninstallBatch else {
                throw TrashEmptyingCheckFailure(description: "cleanup fixtures must have executable selections before acquiring the trash gate")
            }
            controller.requestEmptyTrash(model: model)
            controller.requestEmptyTrash(model: model)
            await execution.waitUntilStarted()
            guard controller.isEmptyingTrash, model.trashEmptyingInProgress,
                  !controller.canEmptyTrash(model: model), !model.canRunOneClickCleanup,
                  !model.canExecuteSelection, !model.canExecuteUninstallBatch,
                  await execution.calls == 1, presentation.confirmationCount == 1 else {
                throw TrashEmptyingCheckFailure(description: "pending emptying must block duplicates and every cleanup entry")
            }
            try model.checkTrashEmptyingExecutionGate()
            let delegate = CleanerApplicationDelegate()
            delegate.model = model
            guard !delegate.allowsTermination else {
                throw TrashEmptyingCheckFailure(description: "application termination must stay blocked while emptying is pending")
            }
            let completion = controller.emptyingTask
            await execution.finish(outcome)
            _ = await completion?.value
            await model.waitForTrashEmptyingCapacityCheck()
            guard !controller.isEmptyingTrash, !model.trashEmptyingInProgress,
                  await execution.capacityReads == 1 else {
                throw TrashEmptyingCheckFailure(description: "every completed emptying attempt must refresh capacity and release busy state")
            }
            guard controller.shouldRemind(cleanedBytes: 1_024) == (index != 0),
                  controller.shouldRemind(cleanedBytes: model.sessionCleanupSummary.bytes),
                  presentation.failures.count == (index == 0 ? 0 : 1) else {
                throw TrashEmptyingCheckFailure(description: "only confirmed success may acknowledge captured bytes; failures need visible errors")
            }
        }
        print("PASS: trash success, timeout, operation cancellation, permission denial and unconfirmed completion; nonempty selection gates; cleanup added during operation; capacity refresh")

        preferences.set(true, forKey: "cleanup.suppressTrashReminder")
        let execution = TrashEmptyingFixture()
        let presentation = TrashEmptyingPresentationFixture(approved: true)
        let model = makeTrashEmptyingCheckModel(fixture: execution)
        let controller = TrashReminderController(preferences: preferences,
            confirmEmptyTrash: presentation.confirm, emptyTrash: { try await execution.execute() }, reportFailure: presentation.reportFailure)
        guard controller.remindIfNeeded(model: model) == nil, presentation.confirmationCount == 0 else {
            throw TrashEmptyingCheckFailure(description: "prior persistent suppression must still skip close and quit reminders")
        }
        controller.requestEmptyTrash(model: model)
        await execution.waitUntilStarted()
        let completion = controller.emptyingTask
        await execution.finish(.success(()))
        _ = await completion?.value
        guard preferences.bool(forKey: "cleanup.suppressTrashReminder"), !controller.shouldRemind(cleanedBytes: UInt64.max) else {
            throw TrashEmptyingCheckFailure(description: "successful emptying must preserve prior persistent suppression")
        }
        try FinderTrashEmptying.validate(status: 0, stdout: Data("SUPERCLEANER_TRASH_EMPTY_CONFIRMED\n".utf8), stderr: Data())
        let unconfirmed = Result { try FinderTrashEmptying.validate(status: 0, stdout: Data(), stderr: Data()) }
        let failed = Result { try FinderTrashEmptying.validate(status: 1, stdout: Data("SUPERCLEANER_TRASH_EMPTY_CONFIRMED\n".utf8), stderr: Data("cancelled (-128)".utf8)) }
        guard case .failure = unconfirmed, case .failure = failed else {
            throw TrashEmptyingCheckFailure(description: "process exit alone and a token from a failed process must not confirm emptying")
        }
        print("PASS: trash suppression persists; process success requires exact confirmed-empty token and zero exit status; no Finder commands were sent")
        try await checkTrashExitReuse()
    }

    private static func checkTrashExitReuse() async throws {
        let domain = "SuperCleaner.TrashExitReuse.\(UUID())"
        let preferences = UserDefaults(suiteName: domain)!
        defer { preferences.removePersistentDomain(forName: domain) }
        for quitting in [true, false] {
            let execution = TrashEmptyingFixture()
            let presentation = TrashEmptyingPresentationFixture(approved: false)
            let model = makeTrashEmptyingCheckModel(fixture: execution)
            let controller = TrashReminderController(preferences: preferences,
                confirmEmptyTrash: presentation.confirm, emptyTrash: { try await execution.execute() }, reportFailure: presentation.reportFailure)
            if quitting {
                let delegate = CleanerApplicationDelegate(trashReminder: controller)
                delegate.model = model
                guard delegate.requestTermination(reply: presentation.replyToTermination) == .terminateLater else {
                    throw TrashEmptyingCheckFailure(description: "cancelled quit must reply through the shared confirmation task")
                }
                await delegate.terminationTask?.value
                guard presentation.terminationReplies == [true], delegate.terminationTask == nil else {
                    throw TrashEmptyingCheckFailure(description: "Cancel must preserve Trash and allow explicit application termination")
                }
            } else {
                guard await controller.remindIfNeeded(model: model)?.value == false else {
                    throw TrashEmptyingCheckFailure(description: "Cancel from window closing must cancel emptying")
                }
            }
            guard presentation.confirmationCount == 1, await execution.calls == 0, await execution.capacityReads == 0,
                  !model.trashEmptyingInProgress,
                  controller.shouldRemind(cleanedBytes: model.sessionCleanupSummary.bytes),
                  !preferences.bool(forKey: "cleanup.suppressTrashReminder") else {
                throw TrashEmptyingCheckFailure(description: "Cancel must preserve close and quit reminders without executing Finder or suppressing future reminders")
            }
        }
        for success in [true, false] {
            let execution = TrashEmptyingFixture(holdCapacity: true)
            let presentation = TrashEmptyingPresentationFixture(approved: true)
            let model = makeTrashEmptyingCheckModel(fixture: execution)
            let controller = TrashReminderController(preferences: preferences,
                confirmEmptyTrash: presentation.confirm, emptyTrash: { try await execution.execute() }, reportFailure: presentation.reportFailure)
            let delegate = CleanerApplicationDelegate(trashReminder: controller)
            delegate.model = model
            guard delegate.requestTermination(reply: presentation.replyToTermination) == .terminateLater else {
                throw TrashEmptyingCheckFailure(description: "quit must wait for the shared Trash operation")
            }
            let completion = delegate.terminationTask
            await execution.waitUntilStarted()
            guard presentation.terminationReplies.isEmpty,
                  delegate.requestTermination(reply: presentation.replyToTermination) == .terminateLater,
                  await execution.calls == 1, presentation.confirmationCount == 1 else {
                throw TrashEmptyingCheckFailure(description: "quit must confirm once and block duplicate execution and premature termination")
            }
            await execution.finish(success ? .success(()) : .failure(FinderTrashEmptyingError.unconfirmed))
            await execution.waitUntilCapacityRead()
            guard presentation.terminationReplies.isEmpty, model.trashEmptyingInProgress else {
                throw TrashEmptyingCheckFailure(description: "capacity refresh must complete before replying to quit and releasing the trash gate")
            }
            await execution.finishCapacityRead()
            await completion?.value
            guard presentation.terminationReplies == [true], await execution.capacityReads == 1,
                  !model.trashEmptyingInProgress, delegate.terminationTask == nil,
                  controller.shouldRemind(cleanedBytes: model.sessionCleanupSummary.bytes) == !success,
                  presentation.failures.count == (success ? 0 : 1) else {
                throw TrashEmptyingCheckFailure(description: "quit must proceed after the optional emptying attempt and capacity refresh, including failure; failure must retain reminders")
            }
        }
        preferences.removePersistentDomain(forName: domain)
        let execution = TrashEmptyingFixture()
        let presentation = TrashEmptyingPresentationFixture(approved: true)
        let model = makeTrashEmptyingCheckModel(fixture: execution)
        let controller = TrashReminderController(preferences: preferences,
            confirmEmptyTrash: presentation.confirm, emptyTrash: { try await execution.execute() }, reportFailure: presentation.reportFailure)
        let close = controller.remindIfNeeded(model: model)
        await execution.waitUntilStarted()
        await execution.finish(.success(()))
        guard await close?.value == true, presentation.confirmationCount == 1,
              controller.remindIfNeeded(model: model) == nil else {
            throw TrashEmptyingCheckFailure(description: "successful close-window cleanup must use the shared executor and suppress the following quit reminder")
        }
        print("PASS: close/quit reuse the same two-choice confirmation and executor, allow Cancel to quit without emptying, wait for an approved operation and capacity even on failure, preserve reminders and prevent duplicate termination; no Finder commands")
    }
}
